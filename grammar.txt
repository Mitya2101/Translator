<Start> ::= <Program>

<Program> ::= { <Statement> }

<ProgramNoCreateFunction> ::= { <StatementNoCreationFunction> }

<Statement> ::=
      <Expr> ";"
    | <CreateFunctionOrVariableOrArray>
    | <If>
    | <While>
    | <For>
    | <Return> ";"
    | <Break> ";"
    | <Continue> ";"

<StatementNoCreationFunction> ::=
      <Expr> ";"
    | <CreateVariableOrArray> ";"
    | <If>
    | <While>
    | <For>
    | <Return> ";"
    | <Break> ";"
    | <Continue> ";"

<Return> ::= "return" [<Expr>]
<Break> ::= "break"
<Continue> ::= "continue"

<String> ::= <letter> {<letter>}
<StringWithDigit> ::= {<letter> | <digit>}

<Variable> ::= <letter> {<letter> | <digit>}

<letter> ::= /[a-zA-Z]/

<digit> ::= /[0-9]/

<Value> ::= <digit> { <digit> }

<DoubleValue> ::= <Value> [ "." <Value> ]

<Type> ::= "int" | "bool" | "char" | "double"

<CreateVariableOrArray> ::= <Type> <Variable> ("=" <Expr> | ("[" Expr "]", { "[" Expr "]" }))

<CreateFunctionOrVariableOrArray> ::= <Type> <Variable> 
( "(" <ArgListType> ")" "{" <ProgramNoCreateFunction> "}"  
| "=" <Expr> ";" | ("[" Expr "]", { "[" Expr "]" }) ";");

<If> ::=
    "if" "(" <Expr> ")" "{" <ProgramNoCreateFunction> "}"
    [ "else" ( <If> | <Else> ) ]
<Else> ::= "{" <ProgramNoCreateFunction> "}"

<While> ::= "while" "(" <Expr> ")" "{" <ProgramNoCreateFunction> "}"

<For> ::=
    "for" "("
      [ <CreateVariableOrArray> | <Expr> ] ";" 
      [ <Expr> ] ";" 
      [ <Expr> ]
    ")" "{" <ProgramNoCreateFunction> "}"

<Expr> ::= <ExprComma>

<ExprComma> ::= <ExprAssign> { "," <ExprAssign> }

<ExprAssign> ::= <ExprLogicOr> { "=" <ExprAssign> }

<ExprLogicOr> ::= <ExprLogicAnd> { "||" <ExprLogicAnd> }

<ExprLogicAnd> ::= <ExprEquality> { "&&" <ExprEquality> }

<ExprEquality> ::= <ExprRel> { ( "==" | "!=" ) <ExprRel> }

<ExprRel> ::= <ExprAdd> { ( ">" | "<" | ">=" | "<=" ) <ExprAdd> }

<ExprAdd> ::= <ExprMul> { ( "+" | "-" ) <ExprMul> }

<ExprMul> ::= <ExprPostfix> { ( "*" | "/" | "%" ) <ExprPostfix> }

<ExprPostfix> ::= <ExprUnary> ("(" [<ArgList>] ")" | {"[" <Expr> "]"} | )

<ExprUnary> ::= ["+" | "-" | "!"] <ExprPrimary>

<ExprPrimary> ::=
     <StringWithDigit> 
    |  <Variable>
    | <DoubleValue>
    | "(" <Expr> ")"

<ArgList> ::= <ExprAssign> { "," <ExprAssign> }
<ArgListType> ::= <Type> <Variable> { "," <Type> <Variable> }
