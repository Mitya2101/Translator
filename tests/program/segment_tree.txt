int t[4 * 8];
int a[8];
a[0] = 5;
a[1] = 4;
a[2] = 2;
a[3] = 1;
a[4] = 7;
a[5] = -1;
a[6] = 0;
a[7] = 1;

void build(int i,int l,int r){
    if(l == r - 1){
        t[i] = a[l];
        return;
    }
    int mid = (l + r) / 2;
    build(2 * i + 1,l,mid);
    build(2 * i + 2,mid,r);
    
    if(t[2 * i + 1] > t[2 * i + 2]){
        t[i] = t[2 * i + 1];
    }else{
        t[i] = t[2 * i + 2];
    }
}

int query(int i,int l,int r,int ql,int qr){
    if(l >= ql && r <= qr){
        return t[i];
    }
    if(l >= qr || ql >= r){
        return 0;
    }
    int mid = (l + r) / 2;
    int left = query(2 * i + 1,l,mid,ql,qr);
    int right = query(2 * i + 2,mid,r,ql,qr);
    if(left > right){
        return left;
    }
    return right;
}

void upd(int i,int l,int r,int pos,int x){
    if(l == r - 1){
        t[i] = x;
        return;
    }
    int mid = (l + r) / 2;
    if(pos >= mid){
        upd(2 * i + 2,mid,r,pos,x);
    }else{
        upd(2 * i + 1,l,mid,pos,x);
    }
    if(t[2 * i + 1] > t[2 * i + 2]){
        t[i] = t[2 * i + 1];
    }else{
        t[i] = t[2 * i + 2];
    }
}

build(0,0,8);
print(query(0,0,8,5,8));
print(query(0,0,8,0,2));
upd(0,0,8,1,10);
print(query(0,0,8,0,2));
print(query(0,0,8,1,4));
upd(0,0,8,1,0);
print(query(0,0,8,0,2));
print(query(0,0,8,1,4));
print(query(0,0,8,0,8));
